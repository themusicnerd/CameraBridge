cmake_minimum_required(VERSION 3.21)
project(CameraBridgeSony LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT SONY_SAMPLE_ROOT)
  message(FATAL_ERROR "Set SONY_SAMPLE_ROOT to the extracted RemoteCli directory")
endif()
add_executable(CameraBridge-Sony main.cpp)
target_include_directories(CameraBridge-Sony PRIVATE "${SONY_SAMPLE_ROOT}/app/CRSDK")
find_library(SONY_CORE NAMES Cr_Core PATHS "${SONY_SAMPLE_ROOT}/external/crsdk" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(CameraBridge-Sony PRIVATE "${SONY_CORE}")
if(WIN32)
  target_compile_definitions(CameraBridge-Sony PRIVATE UNICODE _UNICODE)
else()
  target_link_libraries(CameraBridge-Sony PRIVATE pthread)
  set_target_properties(CameraBridge-Sony PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
add_custom_command(TARGET CameraBridge-Sony POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${SONY_SAMPLE_ROOT}/external/crsdk" "$<TARGET_FILE_DIR:CameraBridge-Sony>")
